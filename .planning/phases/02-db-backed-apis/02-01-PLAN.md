**Service file (src/modules/programs/programs.service.ts):**

Full replacement. No imports of HttpService, firstValueFrom, or OPINTOPOLKU_BASE should remain.

```typescript
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../../providers/prisma.service';

@Injectable()
export class ProgramsService {
  constructor(private readonly prisma: PrismaService) {}

  async findAll(query: Record<string, any> = {}) {
    const size = Number(query.size ?? 20);
    const page = Number(query.page ?? 0);

    const [total, rows] = await this.prisma.$transaction([
      this.prisma.program.count(),
      this.prisma.program.findMany({
        skip: page * size,
        take: size,
        include: {
          universities: {
            include: { university: { select: { oid: true, name: true } } },
          },
        },
      }),
    ]);

    return {
      total,
      page,
      size,
      hits: rows.map((row) => this.mapProgram(row)),
    };
  }

  async findOne(oid: string, _lng?: string) {
    const program = await this.prisma.program.findUnique({
      where: { oid },
      include: {
        universities: {
          include: { university: { select: { oid: true, name: true } } },
        },
      },
    });
    if (!program) throw new NotFoundException(`Program not found: ${oid}`);
    return this.mapProgramDetail(program);
  }

  private mapProgram(row: any) {
    return {
      oid: row.oid,
      name: row.name,
      type: row.type,
      isDegree: row.isDegree,
      imageUrl: row.imageUrl ?? null,
      fieldOfStudy: row.fieldOfStudy ?? null,
      creditsAmount: row.creditsAmount ?? null,
      creditsUnit: row.creditsUnit ?? null,
      teachingLanguages: row.teachingLanguages ?? [],
      providers: (row.universities ?? []).map((pu: any) => ({
        oid: pu.university.oid,
        name: pu.university.name,
      })),
    };
  }

  private mapProgramDetail(row: any) {
    return {
      oid: row.oid,
      name: row.name,
      description: row.description ?? null,
      type: row.type,
      typePath: row.typePath ?? null,
      isDegree: row.isDegree,
      imageUrl: row.imageUrl ?? null,
      creditsAmount: row.creditsAmount ?? null,
      creditsUnit: row.creditsUnit ?? null,
      eqfLevel: row.eqfLevel ?? null,
      nqfLevel: row.nqfLevel ?? null,
      fieldOfStudy: row.fieldOfStudy ?? null,
      degreeTitles: row.degreeTitles ?? [],
      teachingLanguages: row.teachingLanguages ?? [],
      implementations: row.implementations ?? null,
      universities: (row.universities ?? []).map((pu: any) => ({
        oid: pu.university.oid,
        name: pu.university.name,
      })),
    };
  }
}
```

Note: The `_lng` parameter is accepted but unused — the DB stores English strings already. This keeps the controller contract intact.

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP client → ProgramsController | Query params `page`, `size` arrive as strings from any caller |
| ProgramsService → PostgreSQL | Prisma queries with parameterized inputs (ORM-managed, no raw SQL) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-02-01 | Tampering | `findAll` pagination — `page` and `size` coerced with `Number()` | mitigate | `Number(undefined)` returns `NaN`; use `?? 0` / `?? 20` defaults before coercion so skip/take never become NaN. Implemented inline in mapProgram. |
| T-02-02 | Denial of Service | Unbounded `size` parameter — caller requests size=100000 | accept | No upper-bound enforced in this phase. Acceptable for v1 with low traffic; throttling added in Phase 5 (SRCH-07). |
| T-02-03 | Information disclosure | findOne 404 exposes OID in message | accept | OIDs are public identifiers from Opintopolku (not secret). 404 message is safe. |
| T-02-04 | Elevation of privilege | `POST /sync/run` is unguarded per D-10 | accept | Sync is read-from-upstream, write-to-own-DB — no user data exposure. Auth hardening deferred to Phase 5. |
</threat_model>
